(fn DirectorTick ()
  (if (Variables|Default|GetEnabled)
    (CallFunction|SyncPawns)
    (for t (range 2)
      (bind team (Math|Conversions|ToByte(Integer) t))
      (bind humans (CallFunction|CountHumans :Team team))
      (bind desired (Math|Integer|Max(Integer) (- (Variables|Default|GetTeamSize) humans) 0))
      (bind have (CallFunction|CountBots :Team team :AliveOnly false))
      (if (< have desired)
        (for i (range (Math|Integer|Min(Integer) (- desired have) (Variables|Default|GetMaxSpawnPerTick)))
          (CallFunction|SpawnBot :Team team))
        (else
          (bind alive (CallFunction|CountBots :Team team :AliveOnly true))
          (if (> alive desired)
            (CallFunction|RemoveOneBot :Team team)))))))
